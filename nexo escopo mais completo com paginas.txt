router.py

import json
from datetime import datetime
from zoneinfo import ZoneInfo
from openai import OpenAI
from config import config
from prompts import ROUTER_SYSTEM
from tools import TOOLS

client = OpenAI(api_key=config.OPENAI_API_KEY)

def rotear(pergunta: str, usuario: str) -> tuple[str, dict]:
    agora = datetime.now(ZoneInfo(config.TIMEZONE))
    system = ROUTER_SYSTEM.format(
        empresa=config.EMPRESA,
        data_atual=agora.strftime("%Y-%m-%d (%A)"),
        usuario=usuario,
    )
    resp = client.chat.completions.create(
        model=config.MODEL_ROUTER,
        messages=[
            {"role": "system", "content": system},
            {"role": "user", "content": pergunta},
        ],
        tools=TOOLS,
        tool_choice="required",   # obriga escolher uma ferramenta
        temperature=0,            # roteamento é decisão, não criatividade
    )
    msg = resp.choices[0].message
    if not msg.tool_calls:
        return "fora_de_escopo", {"motivo": "não foi possível interpretar"}
    call = msg.tool_calls[0]
    return call.function.name, json.loads(call.function.arguments or "{}")

from supabase import create_client
from config import config

supabase = create_client(config.SUPABASE_URL, config.SUPABASE_KEY)

# intents que NÃO chamam banco (tratadas no fluxo)
NAO_DB = {"fora_de_escopo", "pedir_esclarecimento"}

# intents que existem como RPC no Supabase
RPCS = {
    "nexo_vendas_periodo",
    "nexo_comparativo_mes",
    "nexo_comparativo_ano",
    "nexo_ranking_vendedores",
    "nexo_vendedores_abaixo_meta",
    "nexo_top_produtos",
    "nexo_top_clientes",
    "nexo_clientes_inativos",
    "nexo_vendas_regiao",
    "nexo_relatorio_periodo",
}

def executar(intent: str, params: dict):
    if intent not in RPCS:
        raise ValueError(f"Intent '{intent}' não é uma RPC válida.")
    resp = supabase.rpc(intent, params).execute()
    return resp.data

import json
from openai import OpenAI
from config import config
from prompts import FORMATTER_SYSTEM

client = OpenAI(api_key=config.OPENAI_API_KEY)

def formatar(pergunta: str, intent: str, dados) -> str:
    if not dados:
        return "Não encontrei registros para esse período. Confira as datas ou tente outro intervalo."

    prompt = FORMATTER_SYSTEM.format(
        empresa=config.EMPRESA,
        pergunta=pergunta,
        intent=intent,
        dados=json.dumps(dados, ensure_ascii=False, indent=2, default=str),
    )
    resp = client.chat.completions.create(
        model=config.MODEL_FORMATTER,
        messages=[{"role": "user", "content": prompt}],
        temperature=0.3,
    )
    return resp.choices[0].message.content

import json
from datetime import datetime, timezone
from executor import supabase

def log_consulta(usuario: str, pergunta: str, intent: str, params: dict, ok: bool):
    """Auditoria: grava toda consulta na tabela nexo_logs (crie no banco)."""
    try:
        supabase.table("nexo_logs").insert({
            "usuario": usuario,
            "pergunta": pergunta,
            "intent": intent,
            "parametros": json.dumps(params, ensure_ascii=False, default=str),
            "sucesso": ok,
            "created_at": datetime.now(timezone.utc).isoformat(),
        }).execute()
    except Exception:
        pass  # log nunca pode derrubar a resposta

main.py

from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel, Field

from router import rotear
from executor import executar, NAO_DB
from formatter import formatar
from logger import log_consulta

app = FastAPI(title="NEXO API", version="1.0")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],          # em produção: restrinja ao domínio do front
    allow_methods=["POST", "GET"],
    allow_headers=["*"],
)

# ---------- Schemas ----------
class PerguntaIn(BaseModel):
    pergunta: str = Field(..., min_length=2, max_length=500)
    usuario: str = Field(default="anonimo")

class RespostaOut(BaseModel):
    tipo: str            # "analise" | "esclarecimento" | "fora_escopo"
    conteudo: str        # texto final pro usuário
    intent: str | None = None
    dados: object | None = None   # dados crus (front pode montar gráfico)

# ---------- Endpoint principal ----------
@app.post("/nexo/perguntar", response_model=RespostaOut)
def perguntar(body: PerguntaIn):
    pergunta, usuario = body.pergunta.strip(), body.usuario

    # 1. Roteamento (LLM barato decide a intenção)
    try:
        intent, params = rotear(pergunta, usuario)
    except Exception as e:
        raise HTTPException(502, f"Falha no roteador: {e}")

    # 2. Casos que não vão ao banco
    if intent == "fora_de_escopo":
        log_consulta(usuario, pergunta, intent, params, True)
        return RespostaOut(
            tipo="fora_escopo",
            conteudo="Ainda não sei responder isso. Posso ajudar com "
                     "vendas, faturamento, clientes, produtos, vendedores e metas.",
            intent=intent,
        )

    if intent == "pedir_esclarecimento":
        log_consulta(usuario, pergunta, intent, params, True)
        return RespostaOut(
            tipo="esclarecimento",
            conteudo=params.get("pergunta_ao_usuario",
                                 "Pode me dizer o período que você quer analisar?"),
            intent=intent,
        )

    # 3. Executa a RPC (banco calcula o número REAL)
    try:
        dados = executar(intent, params)
    except Exception as e:
        log_consulta(usuario, pergunta, intent, params, False)
        raise HTTPException(500, f"Erro ao consultar dados: {e}")

    log_consulta(usuario, pergunta, intent, params, True)

    # 4. Formatação (LLM bom escreve a análise — sem calcular)
    try:
        texto = formatar(pergunta, intent, dados)
    except Exception:
        texto = "Consegui os dados, mas falhei ao formatar a análise."

    return RespostaOut(tipo="analise", conteudo=texto, intent=intent, dados=dados)

# ---------- Health ----------
@app.get("/health")
def health():
    return {"status": "ok", "service": "NEXO API"}

NEXO — Resumo do Projeto
A visão em uma frase
NEXO é a camada de inteligência conversacional que conecta os sistemas de uma empresa e permite que qualquer gestor pergunte, em texto ou voz, o que quiser sobre seu negócio — e receba a resposta em análise, gráfico ou relatório, na hora.

Fluxo do produto (do contrato ao uso)


1. CONTRATO
   Empresa fecha com NEXO. Requisito: acesso ao banco/sistemas via API.
        ▼
2. CONEXÃO & TESTES
   Você conecta as fontes de dados, cria as RPCs/consultas e valida
   números contra os relatórios oficiais da empresa (bate 100%).
        ▼
3. ONBOARDING DO USUÁRIO
   Baixa o PWA (celular ou desktop). Cadastro próprio:
   CNPJ + setor + e-mail + senha. NEXO lê os dados do contrato
   e provisiona o acesso do cliente.
        ▼
4. USO DIÁRIO
   Tela simples (estilo ChatGPT/Claude): input de texto e áudio.
   Usuário pergunta → NEXO interpreta → converte em consulta →
   busca nos sistemas da empresa → responde com análise + dashboards
   interativos, gráficos e relatórios.
        ▼
5. FUTURO (fase 2)
   NEXO capta dados externos da internet: mercado, nicho do cliente,
   tendências, insights — sempre com fontes atualizadas e confiáveis.
O que já está construído (fundação técnica)
✅ 10 RPCs no Supabase (vendas, comparativos, rankings, metas, top produtos/clientes, inativos, regiões, relatório consolidado)
✅ Backend FastAPI completo com arquitetura de 3 camadas:
Roteador (LLM barato, temp=0) → escolhe a ferramenta certa
Executor → banco calcula o número real (IA nunca inventa)
Formatter (LLM bom) → escreve a análise em PT-BR
✅ Auditoria (nexo_logs) de toda pergunta
✅ Regras de negócio definidas: faturamento por data_faturamento, status atendido/parcialmente atendido, meta mensal proporcional ao período
Minha opinião honesta
A tese está certa e o timing é bom. Você identificou uma dor real: o gestor tem Power BI, apps e relatórios — mas todos exigem que ele saiba onde procurar e como montar. NEXO inverte isso: o gestor pergunta em linguagem natural e a inteligência faz o trabalho. Isso é, de fato, para onde os sistemas empresariais estão indo. Sua leitura do futuro está alinhada com o mercado.

O diferencial mais forte do NEXO não é a IA — é a confiabilidade. Sua arquitetura onde o banco calcula e a IA só narra resolve o maior medo de gestor com IA: número inventado. Venda isso como o principal argumento. "Outros assistentes chutam. O NEXO só fala o que o seu banco confirma."

Onde focar (e onde ter cuidado)



Ponto forte	Aproveite
Multi-sistema num só lugar	É o nome do produto — orquestração. Cada cliente conecta ERP, e-commerce, financeiro
Voz + texto	Gestor no carro/fábrica pergunta falando. Poucos concorrentes fazem bem
Confiabilidade	Seu maior trunfo de venda



Cuidado	Por quê
Multi-tenant / isolamento	O CNPJ define o "mundo" de cada cliente. Um cliente NUNCA pode ver dado de outro. Isso é vida ou morte do produto — priorize RLS/isolamento por tenant desde já
Cada empresa tem schema diferente	As 10 RPCs foram feitas pro schema da Designtex. Você vai precisar de um processo de onboarding técnico (mapear tabelas → gerar RPCs) pra cada novo cliente. Isso é seu gargalo de escala — pense em como semi-automatizar
LGPD	Dados empresariais + IA + fontes externas. Contrato precisa cobrir tratamento, retenção e onde os dados trafegam (as chamadas ao OpenAI saem do país)
Permissão por perfil	Vendedor vê só o dele; gestor vê tudo. Já mapeamos isso como próximo passo — é essencial pra vender pra empresa inteira, não só pro dono
Sugestões de melhorias (roadmap)
Curto prazo (v1 sólida):

Isolamento multi-tenant por CNPJ (o mais crítico)
Perfis de acesso (gestor / vendedor / setor)
Dashboards interativos — o front consome os dados crus que a API já devolve e monta gráfico
Memória de conversa ("e no mês passado?")
Alertas proativos — NEXO avisa sem ser perguntado: "queda de 15% na região Sul esta semana"
Médio prazo (diferenciação):

Relatório agendado — toda segunda 8h, resumo no e-mail/WhatsApp
Onboarding assistido por IA — a própria IA lê o schema do novo cliente e sugere as consultas (ataca seu gargalo de escala)
Export PDF/Excel dos relatórios

NEXO — Isolamento Multi-Tenant por CNPJ
Este é o alicerce que permite atender N empresas com zero risco de vazamento entre elas. A regra é absoluta: o CNPJ define o mundo do usuário, e nada atravessa essa fronteira.

Conceito: 3 camadas de defesa


┌─────────────────────────────────────────────┐
│  CAMADA 1: Autenticação                       │
│  Login → token contém tenant_id (o CNPJ)      │
├─────────────────────────────────────────────┤
│  CAMADA 2: Backend (defesa em profundidade)   │
│  Toda RPC recebe o tenant do TOKEN,           │
│  NUNCA do que o usuário digitou               │
├─────────────────────────────────────────────┤
│  CAMADA 3: Banco (RLS - última barreira)      │
│  Mesmo com bug no código, o banco recusa      │
│  dado de outro tenant                         │
└─────────────────────────────────────────────┘
Por que 3 camadas: se uma falhar, as outras seguram. Isolamento é vida ou morte do produto — não confie em uma só.

Passo 2 — Row Level Security (a barreira do banco)
sql


-- Ativa RLS
alter table pedidos_totvs enable row level security;
alter table clientes      enable row level security;
alter table representantes enable row level security;

-- Função que lê o tenant do usuário logado
create or replace function meu_tenant()
returns uuid language sql stable security definer as $$
  select tenant_id from nexo_usuarios
  where auth_user_id = auth.uid();
$$;

-- Policy: usuário só enxerga linhas do próprio tenant
create policy tenant_isolation on pedidos_totvs
  for select using (tenant_id = meu_tenant());

create policy tenant_isolation on clientes
  for select using (tenant_id = meu_tenant());

create policy tenant_isolation on representantes
  for select using (tenant_id = meu_tenant());
Com RLS ligado, mesmo que o backend tenha um bug, o Postgres devolve zero linhas de outro tenant. Essa é a rede de segurança final.

Passo 3 — RPCs passam a receber o tenant
As 10 RPCs ganham p_tenant uuid e filtram por ele. Exemplo na nº 1:

sql


create or replace function nexo_vendas_periodo(
  p_tenant uuid, inicio date, fim date
)
returns table(pedidos bigint, total numeric, ticket numeric)
language sql stable security definer as $$
  select count(*), coalesce(sum(valor_total),0), coalesce(avg(valor_total),0)
  from pedidos_totvs
  where tenant_id = p_tenant                       -- 🔒 isolamento
    and data_faturamento::date between inicio and fim
    and lower(status) in ('atendido','parcialmente atendido');
$$;
E o filtro por vendedor (perfil vendedor vê só o dele):

sql


create or replace function nexo_ranking_vendedores(
  p_tenant uuid, p_rep text, inicio date, fim date, limite int default 10
)
returns table(representante_codigo text, representante_nome text, total numeric, pedidos bigint)
language sql stable security definer as $$
  select representante_codigo, max(representante_nome),
         coalesce(sum(valor_total),0), count(*)
  from pedidos_totvs
  where tenant_id = p_tenant
    and (p_rep is null or representante_codigo = p_rep)  -- 🔒 vendedor vê só o dele
    and data_faturamento::date between inicio and fim
    and lower(status) in ('atendido','parcialmente atendido')
  group by representante_codigo
  order by 3 desc
  limit limite;
$$;

from fastapi import Header, HTTPException

def get_contexto(authorization: str = Header(...)) -> dict:
    """Extrai o usuário do token e devolve seu tenant/perfil.
    O tenant NUNCA vem do corpo da requisição — só do token validado."""
    token = authorization.replace("Bearer ", "")
    try:
        user = supabase.auth.get_user(token).user
    except Exception:
        raise HTTPException(401, "Token inválido.")

    resp = (supabase.table("nexo_usuarios")
            .select("tenant_id, perfil, representante_codigo, ativo, email")
            .eq("auth_user_id", user.id).single().execute())

    u = resp.data
    if not u or not u["ativo"]:
        raise HTTPException(403, "Usuário sem acesso.")

    return {
        "tenant_id": u["tenant_id"],
        "perfil": u["perfil"],
        "rep": u["representante_codigo"] if u["perfil"] == "vendedor" else None,
        "email": u["email"],
    }

def executar(intent: str, params: dict, ctx: dict):
    if intent not in RPCS:
        raise ValueError(f"Intent '{intent}' inválida.")

    # 🔒 injeta SEMPRE do contexto do token, não do LLM
    params = dict(params)
    params["p_tenant"] = ctx["tenant_id"]

    # RPCs que respeitam filtro por vendedor
    if intent in {"nexo_ranking_vendedores", "nexo_vendas_periodo",
                  "nexo_top_clientes", "nexo_top_produtos",
                  "nexo_vendas_regiao"}:
        params["p_rep"] = ctx["rep"]   # None = vê tudo (gestor)

    return supabase.rpc(intent, params).execute().data

Etapa 1 — IA analisa o schema e propõe o mapeamento
Antes de gerar SQL, a IA precisa entender o banco do cliente: qual tabela é a fonte da verdade de vendas, qual campo é data de faturamento, etc.

import json
from openai import OpenAI
from config import config

client = OpenAI(api_key=config.OPENAI_API_KEY)

ANALISTA_SYSTEM = """Você é o engenheiro de dados do NEXO. Recebe o schema \
de um banco de um cliente novo e produz um MAPEAMENTO SEMÂNTICO: identifica \
quais tabelas/colunas correspondem aos conceitos que o NEXO precisa.

Conceitos que o NEXO precisa localizar:
- fonte_vendas: tabela principal de pedidos/vendas faturadas
- data_faturamento: coluna de data em que a venda virou receita
- data_pedido: coluna de data em que o pedido entrou
- valor_venda: coluna de valor total do pedido/venda
- status: coluna de situação + quais valores significam "faturado/válido"
- cliente: tabela e chave de clientes
- representante: tabela e chave de vendedores + coluna de meta (se houver)
- itens: onde estão os itens (coluna JSON ou tabela filha) e as chaves de
  produto, quantidade faturada e preço
- regiao: coluna de UF/estado

REGRAS:
- Baseie-se SÓ no schema fornecido. Não invente colunas.
- Para cada conceito, dê: tabela, coluna e um campo "confianca" (alta/media/baixa).
- Se um conceito NÃO existir no schema, marque como "ausente" e explique o impacto.
- Liste "perguntas_ao_integrador": dúvidas que só um humano do cliente resolve
  (ex.: quais status contam como faturado).
- Responda SOMENTE em JSON válido.
"""

def analisar_schema(schema_texto: str) -> dict:
    resp = client.chat.completions.create(
        model=config.MODEL_FORMATTER,   # tarefa complexa → modelo bom
        messages=[
            {"role": "system", "content": ANALISTA_SYSTEM},
            {"role": "user", "content": f"SCHEMA DO CLIENTE:\n{schema_texto}"},
        ],
        temperature=0,
        response_format={"type": "json_object"},
    )
    return json.loads(resp.choices[0].message.content)

GERADOR_SYSTEM = """Você é o gerador de RPCs do NEXO. Recebe um MAPEAMENTO \
SEMÂNTICO já validado por um humano e gera as 10 funções SQL (PostgreSQL/Supabase) \
padrão do NEXO, adaptadas às tabelas e colunas REAIS daquele mapeamento.

As 10 RPCs (mesma assinatura padrão do NEXO):
1. nexo_vendas_periodo(p_tenant, inicio, fim)
2. nexo_comparativo_mes(p_tenant, mes_ref)
3. nexo_comparativo_ano(p_tenant, inicio, fim)
4. nexo_ranking_vendedores(p_tenant, p_rep, inicio, fim, limite)
5. nexo_vendedores_abaixo_meta(p_tenant, inicio, fim)
6. nexo_top_produtos(p_tenant, inicio, fim, limite)
7. nexo_top_clientes(p_tenant, p_rep, inicio, fim, limite)
8. nexo_clientes_inativos(p_tenant, dias_sem_comprar)
9. nexo_vendas_regiao(p_tenant, inicio, fim)
10. nexo_relatorio_periodo(p_tenant, inicio, fim)

REGRAS INEGOCIÁVEIS:
- Use APENAS as tabelas/colunas do mapeamento. NUNCA invente nomes.
- TODA RPC filtra por tenant_id = p_tenant (isolamento multi-tenant).
- Vendas por data_faturamento; status conforme os valores validados.
- Todas 'security definer' e 'stable'.
- Se um conceito estava 'ausente' no mapeamento, gere a RPC mesmo assim mas
  adicione um comentário SQL "-- ⚠️ AJUSTAR: <motivo>" na linha afetada.
- Para itens em JSON, use jsonb_array_elements com as chaves reais mapeadas.
- Para itens em tabela filha, use JOIN.
- Devolva JSON: {"rpcs": [{"nome","sql","observacoes"}], "avisos": [...]}
"""

def gerar_rpcs(mapeamento: dict) -> dict:
    resp = client.chat.completions.create(
        model=config.MODEL_FORMATTER,
        messages=[
            {"role": "system", "content": GERADOR_SYSTEM},
            {"role": "user", "content": "MAPEAMENTO VALIDADO:\n" +
             json.dumps(mapeamento, ensure_ascii=False, indent=2)},
        ],
        temperature=0,
        response_format={"type": "json_object"},
    )
    return json.loads(resp.choices[0].message.content)

from fastapi import APIRouter, Depends, HTTPException, Header
from pydantic import BaseModel
from config import config
from onboarding.analista import analisar_schema
from onboarding.gerador import gerar_rpcs
from executor import supabase

router = APIRouter(prefix="/nexo/onboarding", tags=["onboarding"])

def somente_admin(x_admin_key: str = Header(...)):
    if x_admin_key != config.ADMIN_KEY:      # chave secreta só sua
        raise HTTPException(403, "Acesso restrito ao admin NEXO.")

# ---------- 1. Analisar schema ----------
class SchemaIn(BaseModel):
    cnpj: str
    schema_texto: str     # o SQL/CSV do schema, como você colou da Designtex

@router.post("/analisar", dependencies=[Depends(somente_admin)])
def analisar(body: SchemaIn):
    mapa = analisar_schema(body.schema_texto)
    # guarda pra revisão (não gera nada ainda)
    supabase.table("onboarding_mapeamentos").upsert({
        "cnpj": body.cnpj, "mapeamento": mapa, "status": "aguardando_revisao"
    }, on_conflict="cnpj").execute()
    return {"cnpj": body.cnpj, "mapeamento": mapa,
            "proximo_passo": "Revise, responda as perguntas_ao_integrador e "
                             "chame /gerar com o mapeamento corrigido."}

# ---------- 2. Gerar RPCs (após revisão humana) ----------
class GerarIn(BaseModel):
    cnpj: str
    mapeamento_revisado: dict    # você cola o mapa já corrigido

@router.post("/gerar", dependencies=[Depends(somente_admin)])
def gerar(body: GerarIn):
    resultado = gerar_rpcs(body.mapeamento_revisado)
    supabase.table("onboarding_mapeamentos").update({
        "mapeamento": body.mapeamento_revisado,
        "rpcs_geradas": resultado,
        "status": "rpcs_geradas"
    }).eq("cnpj", body.cnpj).execute()
    return {
        "cnpj": body.cnpj,
        "rpcs": resultado["rpcs"],
        "avisos": resultado.get("avisos", []),
        "proximo_passo": "⚠️ REVISE o SQL. Aplique manualmente no banco do "
                         "cliente. Depois teste contra um relatório oficial."
    }

import { useState, useRef, useEffect } from "react";
import { perguntar, transcrever } from "../api";
import Message from "./Message";

export default function Chat({ token }) {
  const [msgs, setMsgs] = useState([
    { role: "nexo", conteudo: "Olá! Sou o NEXO. Pergunte sobre vendas, clientes, produtos ou metas — por texto ou áudio. 🎙️" },
  ]);
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);
  const [gravando, setGravando] = useState(false);
  const recRef = useRef(null);
  const fimRef = useRef(null);

  useEffect(() => fimRef.current?.scrollIntoView({ behavior: "smooth" }), [msgs]);

  async function enviar(texto) {
    const t = (texto ?? input).trim();
    if (!t || loading) return;
    setInput("");
    setMsgs((m) => [...m, { role: "user", conteudo: t }]);
    setLoading(true);
    try {
      const r = await perguntar(t, token);
      setMsgs((m) => [...m, { role: "nexo", ...r }]);
    } catch (e) {
      setMsgs((m) => [...m, { role: "nexo", conteudo: `⚠️ ${e.message}` }]);
    } finally {
      setLoading(false);
    }
  }

  async function toggleAudio() {
    if (gravando) {
      recRef.current?.stop();
      setGravando(false);
      return;
    }
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    const rec = new MediaRecorder(stream, { mimeType: "audio/webm" });
    const chunks = [];
    rec.ondataavailable = (e) => chunks.push(e.data);
    rec.onstop = async () => {
      stream.getTracks().forEach((t) => t.stop());
      const blob = new Blob(chunks, { type: "audio/webm" });
      setLoading(true);
      try {
        const texto = await transcrever(blob, token);
        await enviar(texto);
      } catch (e) {
        setMsgs((m) => [...m, { role: "nexo", conteudo: `⚠️ ${e.message}` }]);
        setLoading(false);
      }
    };
    recRef.current = rec;
    rec.start();
    setGravando(true);
  }

  return (
    <div className="flex flex-col h-screen bg-slate-900">
      <header className="p-4 text-sky-300 font-bold text-lg border-b border-slate-800">
        NEXO
      </header>

      <div className="flex-1 overflow-y-auto p-4">
        {msgs.map((m, i) => <Message key={i} msg={m} />)}
        {loading && <div className="text-slate-500 text-sm">NEXO está pensando…</div>}
        <div ref={fimRef} />
      </div>

      <div className="p-3 border-t border-slate-800 flex gap-2">
        <input
          className="flex-1 bg-slate-800 text-white rounded-full px-4 py-2 outline-none"
          placeholder="Pergunte ao NEXO…"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => e.key === "Enter" && enviar()}
        />
        <button onClick={toggleAudio}
          className={`rounded-full w-11 h-11 ${gravando ? "bg-red-500 animate-pulse" : "bg-slate-700"}`}>
          🎙️
        </button>
        <button onClick={() => enviar()}
          className="bg-sky-600 rounded-full px-5 text-white font-medium">
          ➤
        </button>
      </div>
    </div>
  );
}

export async function perguntarStream(pergunta, historico, token, onDados, onDelta, onFim) {
  const r = await fetch(`${BASE}/nexo/perguntar-stream`, {
    method: "POST",
    headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}` },
    body: JSON.stringify({ pergunta, historico }),
  });
  const reader = r.body.getReader();
  const dec = new TextDecoder();
  let buf = "";
  while (true) {
    const { value, done } = await reader.read();
    if (done) break;
    buf += dec.decode(value, { stream: true });
    const linhas = buf.split("\n\n");
    buf = linhas.pop();
    for (const l of linhas) {
      if (!l.startsWith("data: ")) continue;
      const ev = JSON.parse(l.slice(6));
      if (ev.tipo === "dados") onDados(ev.intent, ev.dados);
      else if (ev.tipo === "texto") onDelta(ev.delta);
      else if (ev.tipo === "fim") onFim();
    }
  }
}

def rotear(pergunta: str, usuario: str, historico: list | None = None):
    agora = datetime.now(ZoneInfo(config.TIMEZONE))
    system = ROUTER_SYSTEM.format(
        empresa=config.EMPRESA,
        data_atual=agora.strftime("%Y-%m-%d (%A)"),
        usuario=usuario,
    )
    msgs = [{"role": "system", "content": system}]

    # só as últimas 4 trocas → contexto sem estourar token
    for t in (historico or [])[-4:]:
        papel = "user" if t["role"] == "user" else "assistant"
        msgs.append({"role": papel, "content": t["conteudo"]})

    msgs.append({"role": "user", "content": pergunta})

    resp = client.chat.completions.create(
        model=config.MODEL_ROUTER, messages=msgs,
        tools=TOOLS, tool_choice="required", temperature=0,
    )
    call = resp.choices[0].message.tool_calls
    if not call:
        return "fora_de_escopo", {"motivo": "não interpretado"}
    return call[0].function.name, json.loads(call[0].function.arguments or "{}")

RPC de detecção — SQL

create or replace function nexo_detectar_alertas(p_tenant uuid)
returns table(tipo text, severidade text, mensagem text, dados jsonb)
language plpgsql stable security definer as $$
declare
  fat_hoje numeric; fat_ontem numeric;
begin
  -- 1. Queda de faturamento vs semana anterior
  select coalesce(sum(valor_total),0) into fat_hoje from pedidos_totvs
   where tenant_id = p_tenant and data_faturamento::date >= current_date - 7;
  select coalesce(sum(valor_total),0) into fat_ontem from pedidos_totvs
   where tenant_id = p_tenant
     and data_faturamento::date between current_date - 14 and current_date - 8;

  if fat_ontem > 0 and fat_hoje < fat_ontem * 0.85 then
    tipo := 'queda_faturamento'; severidade := 'alta';
    mensagem := format('Faturamento caiu %s%% nesta semana vs a anterior.',
                       round((1 - fat_hoje/fat_ontem) * 100));
    dados := jsonb_build_object('atual', fat_hoje, 'anterior', fat_ontem);
    return next;
  end if;

  -- 2. Clientes que sumiram (compravam e pararam há 60 dias)
  return query
  select 'cliente_inativo', 'media',
         format('%s: sem comprar há mais de 60 dias.', c.nome),
         jsonb_build_object('cliente', c.nome)
  from clientes c
  where c.tenant_id = p_tenant
    and exists (select 1 from pedidos_totvs p where p.cliente_codigo = c.codigo
                and p.tenant_id = p_tenant
                and p.data_faturamento::date between current_date - 180 and current_date - 60)
    and not exists (select 1 from pedidos_totvs p where p.cliente_codigo = c.codigo
                and p.tenant_id = p_tenant
                and p.data_faturamento::date > current_date - 60)
  limit 5;
end;

from executor import supabase
from datetime import datetime, timezone

def rodar_alertas():
    tenants = supabase.table("tenants").select("id").eq("ativo", True).execute().data
    for t in tenants:

if __name__ == "__main__":

NEXO — Camada de Inteligência de Mercado (Fase 2)
Aqui o NEXO deixa de olhar só pra dentro e passa a cruzar o dado interno do cliente com o mundo lá fora. Sai de "quanto vendi" para "por que vendi isso, e o que o mercado está fazendo".

O princípio que muda tudo
Dado interno = verdade do banco (RPCs). Dado externo = mundo (web search com fonte). O NEXO cruza os dois e gera insight — sempre citando a fonte externa, nunca inventando número de mercado.

Passo 3 — Motor de inteligência de mercado
Usa web search com fontes, filtra confiabilidade e devolve com citação.

mercado/motor.py
python


import json
from datetime import datetime
from zoneinfo import ZoneInfo
from openai import OpenAI
from config import config
from executor import supabase, executar

client = OpenAI(api_key=config.OPENAI_API_KEY)

# fontes preferenciais por confiabilidade (ajustável por setor)
FONTES_CONFIAVEIS = [
    "cepea.esalq.usp.br", "ibge.gov.br", "gov.br", "abit.org.br",
    "valor.globo.com", "reuters.com", "bloomberg.com", "canalrural.com.br",
]

MERCADO_SYSTEM = """Você é o analista de mercado do NEXO para a {empresa}.
Setor: {setor} | Nicho: {nicho} | Insumos-chave: {insumos} | Região: {regiao}.
Data atual: {data}.

Você recebe: (1) resultados de pesquisa web recentes e (2), quando houver,
DADOS INTERNOS reais da empresa (fonte da verdade do banco).

REGRAS INEGOCIÁVEIS:
- Sobre o NEGÓCIO da empresa (vendas, produtos), use APENAS os dados internos.
  Nunca invente números internos.
- Sobre o MERCADO, use apenas as fontes da pesquisa. SEMPRE cite a fonte e a
  data entre parênteses. Se a informação não tiver fonte confiável, diga que
  não encontrou dado confiável — não chute.
- Quando houver os dois, CRUZE: explique como o cenário externo se relaciona
  com o desempenho interno, e dê 1-2 recomendações acionáveis.
- Seja executivo, curto, em PT-BR. Valores em R$ 1.234,56.
- Nunca apresente dado de mercado sem fonte.
"""

def _perfil(tenant_id: str) -> dict:
    t = (supabase.table("tenants").select("perfil_mercado")
         .eq("id", tenant_id).single().execute()).data
    return t.get("perfil_mercado") or {}

def analisar_mercado(params: dict, ctx: dict) -> dict:
    p = _perfil(ctx["tenant_id"])
    agora = datetime.now(ZoneInfo(config.TIMEZONE))

    # 1. Pesquisa web (a IA busca com contexto do nicho)
    query = f"{params['assunto']} {p.get('nicho','')} {p.get('regiao_atuacao','Brasil')} 2026"
    resultados = _buscar_web(query)

    # 2. Dado interno pra cruzar (opcional)
    interno = None
    if params.get("cruzar_com_interno") and params.get("intent_interno"):
        try:
            interno = executar(params["intent_interno"], {
                "inicio": params.get("inicio"), "fim": params.get("fim"),
            }, ctx)
        except Exception:
            interno = None

    # 3. Síntese com fontes
    system = MERCADO_SYSTEM.format(
        empresa=config.EMPRESA, setor=p.get("setor","-"),
        nicho=p.get("nicho","-"), insumos=", ".join(p.get("insumos_chave",[])),
        regiao=p.get("regiao_atuacao","Brasil"),
        data=agora.strftime("%Y-%m-%d"),
    )
    user = f"PERGUNTA: {params['assunto']}\n\nRESULTADOS WEB:\n{json.dumps(resultados, ensure_ascii=False)}"
    if interno:
        user += f"\n\nDADOS INTERNOS (verdade do banco):\n{json.dumps(interno, ensure_ascii=False, default=str)}"

    resp = client.chat.completions.create(
        model=config.MODEL_FORMATTER,
        messages=[{"role":"system","content":system},{"role":"user","content":user}],
        temperature=0.3,
    )
    return {"analise": resp.choices[0].message.content, "fontes": resultados}

# Usa a API de web search do seu provedor (ex.: Tavily, Serper, Bing).
# Exemplo com Tavily (bom custo/qualidade e traz fonte + data):
import httpx
from config import config
from mercado.motor import FONTES_CONFIAVEIS

def _buscar_web(query: str, n: int = 5) -> list[dict]:
    r = httpx.post("https://api.tavily.com/search", json={
        "api_key": config.TAVILY_API_KEY,
        "query": query,
        "search_depth": "advanced",
        "max_results": n,
        "include_domains": FONTES_CONFIAVEIS,  # prioriza fontes confiáveis
    }, timeout=20)
    dados = r.json().get("results", [])
    return [{"titulo": d["title"], "trecho": d["content"][:500],
             "fonte": d["url"], "data": d.get("published_date")} for d in dados]
Coloque _buscar_web em busca.py e importe no motor.py (evita import circular — ajuste conforme sua organização). Adicione TAVILY_API_KEY no config.py.

Passo 5 — Alerta proativo de mercado (bônus)

jobs/alertas_mercado.py

from executor import supabase
from mercado.motor import analisar_mercado
from datetime import datetime, timezone

def rodar_alertas_mercado():
    tenants = supabase.table("tenants").select("id, perfil_mercado")\
              .eq("ativo", True).execute().data
    for t in tenants:
        perfil = t.get("perfil_mercado") or {}
        for insumo in perfil.get("insumos_chave", [])[:3]:
            ctx = {"tenant_id": t["id"], "rep": None}
            res = analisar_mercado(
                {"assunto": f"variação de preço {insumo} últimos 7 dias",
                 "cruzar_com_interno": False}, ctx)
            # heurística simples: só alerta se a IA sinalizar movimento relevante
            if any(k in res["analise"].lower() for k in ["alta", "queda", "subiu", "caiu", "disparou"]):
                supabase.table("nexo_alertas").insert({
                    "tenant_id": t["id"], "tipo": "mercado",
                    "severidade": "media",
                    "mensagem": f"Movimento no insumo '{insumo}': {res['analise'][:180]}",
                    "dados": {"fontes": res["fontes"][:2]}, "lido": False,
                    "created_at": datetime.now(timezone.utc).isoformat(),
                }).execute()
